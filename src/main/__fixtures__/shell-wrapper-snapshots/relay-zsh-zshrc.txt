# Manta relay zsh overlay wrapper
__manta_resolve_user_config_dir() {
  typeset -g _manta_resolved_config_dir="${1:-}"
  while [[ "$_manta_resolved_config_dir" == */ ]]; do _manta_resolved_config_dir="${_manta_resolved_config_dir%/}"; done
  if [[ -z "$_manta_resolved_config_dir" || -f "$_manta_resolved_config_dir/.manta-shell-wrapper" || "$_manta_resolved_config_dir" == */shell-ready/zsh ]]; then
    _manta_resolved_config_dir="$HOME"
  fi
}
__manta_resolve_user_config_dir "${MANTA_USER_ZDOTDIR:-${MANTA_ORIG_ZDOTDIR:-$HOME}}"
_manta_home="$_manta_resolved_config_dir"
if [[ -o interactive && -f "$_manta_home/.zshrc" ]]; then
  _manta_wrapper_zdotdir="$ZDOTDIR"
  # Why: user startup files resolve plugin/config paths from their own ZDOTDIR;
  # Manta restores its wrapper dir afterward so zsh still loads wrapper files.
  export ZDOTDIR="$_manta_home"
  source "$_manta_home/.zshrc"
  export ZDOTDIR="$_manta_wrapper_zdotdir"
  unset _manta_wrapper_zdotdir
fi

if [[ ! -o login ]] || { [[ -o ksharrays || -o shwordsplit || -o shglob ]] 2>/dev/null && [[ "$(emulate 2>/dev/null)" != zsh ]]; }; then
  (( ${+functions[__manta_shell_epilogue]} )) && __manta_shell_epilogue
fi
